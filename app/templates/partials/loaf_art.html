{# A loaf, drawn to suit the recipe's category. Expects `recipe`, or a bare `category` (a RecipeCategory). Optionally `art_class` for the size. Rye is the pan brick Borodinsky comes out as, wheat a scored boule, enriched a plaited loaf, and anything else a slashed batard. No clip paths and no ids: a list of recipes draws a lot of these on one page. #} {%- set kind = (category.value if category is defined else recipe.category.value) -%} {%- set shown = { "rye": "a dark rye brick", "wheat": "a scored round loaf", "enriched": "a plaited loaf", "other": "a slashed loaf", }.get(kind, "a loaf") -%} {#- Joined rather than interpolated, so an unset size class leaves no double space in the attribute. -#} {%- set classes = ["loaf-art", art_class, "loaf-" ~ kind] | select | join(" ") -%}
